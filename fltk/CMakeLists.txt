find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
find_package(OpenGL REQUIRED COMPONENTS OpenGL)
find_package(LibArchive REQUIRED)
pkg_check_modules(FILETYPES REQUIRED IMPORTED_TARGET gio-2.0)
add_library(fsv-native-core STATIC model.cpp)
target_compile_features(fsv-native-core PUBLIC cxx_std_17)
target_link_libraries(fsv-native-core PUBLIC LibArchive::LibArchive PkgConfig::FILETYPES)
add_executable(fsvr main.cpp spatial-view.cpp)
target_compile_features(fsvr PRIVATE cxx_std_17)
target_compile_options(fsvr PRIVATE -Wall -Wextra)
target_link_libraries(fsvr PRIVATE fsv-native-core fltk::gl-shared OpenGL::GL fltk::fltk-shared fltk::images-shared PkgConfig::FONTCONFIG)
include(GNUInstallDirs)
install(TARGETS fsvr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
