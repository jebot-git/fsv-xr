cmake_minimum_required(VERSION 3.21)
project(fsvr VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickWidgets QuickControls2 Quick3D Concurrent)
option(FSV_XR "Build standalone OpenXR support" ON)
if(FSV_XR)
  find_package(Qt6 6.10 REQUIRED COMPONENTS Quick3DXr)
endif()
find_package(LibArchive REQUIRED)
add_library(fsv-core STATIC src/qt/archivebrowser.cpp src/qt/archives.cpp src/qt/archives.h src/qt/filesystem.cpp src/qt/launchcontext.cpp src/qt/launchcontext.h src/qt/filesystemindex.cpp src/qt/trash.cpp src/qt/trash.h src/qt/treelayout.cpp src/qt/branchgeometry.cpp src/qt/branchgeometry.h src/qt/filesystem.h src/qt/gestures.cpp src/qt/gestures.h src/qt/thumbnail.cpp src/qt/thumbnail.h)
target_link_libraries(fsv-core PUBLIC LibArchive::LibArchive Qt6::Core Qt6::Gui Qt6::Quick Qt6::Quick3D Qt6::Concurrent)
target_include_directories(fsv-core PUBLIC src/qt)
add_executable(fsvr src/qt/main.cpp src/qt/domed.cpp src/qt/domed.h src/qt/appearance.cpp src/qt/appearance.h src/qt/resources.qrc)
target_link_libraries(fsvr PRIVATE fsv-core Qt6::Qml Qt6::QuickControls2 Qt6::Quick3D Qt6::QuickWidgets Qt6::Widgets)
target_compile_definitions(fsvr PRIVATE FSV_HAS_XR=$<BOOL:${FSV_XR}>)
if(FSV_XR)
  target_link_libraries(fsvr PRIVATE Qt6::Quick3DXr)
endif()
include(GNUInstallDirs)
install(TARGETS fsvr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/set-dome-default.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME fsvr-dome-default)
install(FILES data/dome-mimeapps.list DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/org.fsvxr.fsvr.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
if(FSV_XR)
  install(FILES data/org.fsvxr.fsvr-openxr.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME dome-default COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/dome-default.py)
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(fsv-tests tests/core.cpp)
  target_link_libraries(fsv-tests PRIVATE fsv-core Qt6::Test)
  add_test(NAME core COMMAND fsv-tests -platform offscreen)
  add_executable(fsv-appearance-tests tests/appearance.cpp src/qt/appearance.cpp src/qt/appearance.h)
  target_include_directories(fsv-appearance-tests PRIVATE src/qt)
  target_link_libraries(fsv-appearance-tests PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME appearance COMMAND fsv-appearance-tests -platform offscreen)
  add_executable(fsv-ui-tests tests/ui.cpp src/qt/resources.qrc)
  target_link_libraries(fsv-ui-tests PRIVATE fsv-core Qt6::Qml Qt6::QuickControls2 Qt6::Quick3D Qt6::Test)
  if(FSV_XR)
    target_link_libraries(fsv-ui-tests PRIVATE Qt6::Quick3DXr)
    target_compile_definitions(fsv-ui-tests PRIVATE FSV_HAS_XR=1)
  endif()
  # Explicit opt-in: Vulkan rendering tests need a live Wayland/X11 display.
  option(FSV_DISPLAY_TESTS "Run rendered UI tests on the current display" OFF)
  if(FSV_DISPLAY_TESTS)
    add_test(NAME ui COMMAND fsv-ui-tests)
  endif()
endif()
